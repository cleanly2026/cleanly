```typescript
import { FastifyInstance } from 'fastify';
import { z } from 'zod';
import prisma from '../../lib/prisma.js';

const querySchema = z.object({
  status: z.enum(['all', 'pending', 'accepted', 'washer_assigned', 'washer_en_route', 'in_progress', 'completed', 'cancelled']).default('all'),
  page: z.coerce.number().int().min(1).default(1),
  limit: z.coerce.number().int().min(1).max(50).default(20),
});

export default async function companyOrdersRoute(fastify: FastifyInstance) {
  // GET /company/orders — paginated order feed for company dashboard (COMP-05)
  fastify.get('/', {
    preHandler: [fastify.authenticate],
    schema: {
      querystring: querySchema,
    },
  }, async (request, reply) => {
    const { sub: userId, companyId } = request.user as { sub: string; companyId: string };
    if (!companyId) return reply.code(403).send({ error: 'Not a company admin' });

    const { status, page, limit } = request.query as z.infer<typeof querySchema>;
    const skip = (page - 1) * limit;

    const where: any = { company_id: companyId };
    if (status !== 'all') where.status = status;

    const [orders, total] = await Promise.all([
      prisma.order.findMany({
        where,
        include: {
          customer: { select: { id: true, first_name: true, last_name: true, phone: true } },
          items: { include: { package: true } },
          washer: { select: { id: true, first_name: true, last_name: true } },
        },
        orderBy: { created_at: 'desc' },
        skip,
        take: limit,
      }),
      prisma.order.count({ where }),
    ]);

    return {
      orders: orders.map(o => ({
        id: o.id,
        order_number: o.order_number,
        customer_name: `${o.customer.first_name} ${o.customer.last_name}`,
        service_category: o.type,
        status: o.status,
        total_amount: o.total_amount,
        created_at: o.created_at,
        washer: o.washer ? { id: o.washer.id, name: `${o.washer.first_name} ${o.washer.last_name}` } : null,
      })),
      pagination: { page, limit, total, pages: Math.ceil(total / limit) },
    };
  });

  // GET /company/washers/available — list washers for assignment dropdown (COMP-06)
  fastify.get('/washers/available', {
    preHandler: [fastify.authenticate],
  }, async (request, reply) => {
    const { companyId } = request.user as { sub: string; companyId: string };
    if (!companyId) return reply.code(403).send({ error: 'Not a company admin' });

    const washers = await prisma.washer.findMany({
      where: { company_id: companyId, is_active: true },
      select: { id: true, first_name: true, last_name: true, status: true },
    });

    return { washers };
  });
}
```

```typescript
import { api } from '../lib/api';

export function useCompanyOrders(status: string, page: number) {
  return useQuery({
    queryKey: ['company-orders', status, page],
    queryFn: () => api.get(`/company/orders?status=${status}&page=${page}&limit=20`).then(r => r.data),

}
```

**Create `apps/company-web/src/hooks/useOrderSocket.ts`:**

Socket.io hook that listens for `order:new` and `order:status-changed` events on the `company:{companyId}` room and invalidates the TanStack Query cache:

```typescript
import { useEffect } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import { socket } from '../lib/socket';

export function useOrderSocket(companyId: string) {
  const queryClient = useQueryClient();

  useEffect(() => {
    socket.emit('join', `company:${companyId}`);

    const handleNewOrder = () => {
      queryClient.invalidateQueries({ queryKey: ['company-orders'] });
    };

    const handleStatusChange = () => {
      queryClient.invalidateQueries({ queryKey: ['company-orders'] });
    };

    socket.on('order:new', handleNewOrder);
    socket.on('order:status-changed', handleStatusChange);

    return () => {
      socket.off('order:new', handleNewOrder);
      socket.off('order:status-changed', handleStatusChange);
      socket.emit('leave', `company:${companyId}`);
    };
  }, [companyId, queryClient]);
}
```

**Create `apps/company-web/src/components/WasherAssignDropdown.tsx`:**

Inline washer assignment per D-14 — dropdown shows available washers, select and confirm without navigating away from the table:

```tsx
import { useState } from 'react';
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { useTranslation } from 'react-i18next';
import { api } from '../lib/api';

export function WasherAssignDropdown({ orderId }: { orderId: string }) {
  const { t } = useTranslation();
  const queryClient = useQueryClient();
  const [isOpen, setIsOpen] = useState(false);
  const [selectedWasherId, setSelectedWasherId] = useState<string | null>(null);

  const { data: washersData } = useQuery({
    queryKey: ['available-washers'],
    queryFn: () => api.get('/company/orders/washers/available').then(r => r.data),
    enabled: isOpen,
  });

  const assignMutation = useMutation({
    mutationFn: (washerId: string) =>
      api.patch(`/orders/${orderId}/assign-washer`, { washer_id: washerId }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['company-orders'] });
      setIsOpen(false);
      setSelectedWasherId(null);
    },
  });

  if (!isOpen) {
    return (
      <button
        onClick={() => setIsOpen(true)}
        className="text-brand-gold font-semibold text-label min-h-[44px]"
      >
        {t('dashboard.assignWasher')}
      </button>
    );
  }

  return (
    <div className="flex items-center gap-sm">
      <select
        value={selectedWasherId || ''}
        onChange={(e) => setSelectedWasherId(e.target.value)}
        className="rounded border border-brand-muted px-sm py-xs text-label min-h-[44px]"
      >
        <option value="">{t('dashboard.selectWasher')}</option>
        {washersData?.washers?.map((w: any) => (
          <option key={w.id} value={w.id}>
            {w.first_name} {w.last_name} — {w.status}
          </option>
        ))}
      </select>
      <button
        onClick={() => selectedWasherId && assignMutation.mutate(selectedWasherId)}
        disabled={!selectedWasherId || assignMutation.isPending}
        className="rounded bg-brand-gold text-white px-sm py-xs text-label min-h-[44px] disabled:opacity-50"
      >
        {t('dashboard.confirm')}
      </button>
    </div>
  );
}
```

```tsx
import { useTranslation } from 'react-i18next';
import { StatusBadge } from './StatusBadge';

interface Order {
  id: string;
  order_number: string;
  customer_name: string;
  service_category: string;
  status: string;
  total_amount: number;
  created_at: string;
  washer: { id: string; name: string } | null;
}

export function OrderTable({ orders, isLoading }: { orders: Order[]; isLoading: boolean }) {
  const { t } = useTranslation();

  if (isLoading) {
    return (
      <div className="space-y-sm">

    return (

    <div className="overflow-x-auto">

                ) : null}
              </td>
```

```tsx
import { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { OrderTable } from '../components/OrderTable';
import { useCompanyOrders } from '../hooks/useCompanyOrders';
import { useOrderSocket } from '../hooks/useOrderSocket';

const TABS = [
  { key: 'all', label: 'dashboard.filterAll' },
  { key: 'pending', label: 'dashboard.filterPending' },
  { key: 'accepted', label: 'dashboard.filterActive' },
  { key: 'completed', label: 'dashboard.filterCompleted' },
];

export function OrderFeed({ companyId }: { companyId: string }) {
  const { t } = useTranslation();
  const [activeTab, setActiveTab] = useState('all');
  const [page, setPage] = useState(1);

  useOrderSocket(companyId);
  const { data, isLoading } = useCompanyOrders(activeTab, page);

  return (
    <div className="p-lg">
      {/* Status filter tabs — 40px height, dense UI per UI-SPEC */}
      <div className="flex gap-md border-b border-brand-muted mb-lg">
        {TABS.map((tab) => (
          <button
            key={tab.key}
            onClick={() => { setActiveTab(tab.key); setPage(1); }}
            className={`h-[40px] px-md text-label transition-colors ${
              activeTab === tab.key
                ? 'font-semibold text-brand-navy border-b-2 border-brand-gold'
                : 'text-brand-navy/60 border-b-2 border-transparent hover:text-brand-navy'
            }`}
          >
            {t(tab.label)}
          </button>
        ))}
      </div>

      <OrderTable orders={data?.orders || []} isLoading={isLoading} />

      {/* Pagination */}
      {data?.pagination && data.pagination.pages > 1 && (
        <div className="flex justify-center gap-sm mt-lg">
          <button
            onClick={() => setPage(p => Math.max(1, p - 1))}
            disabled={page <= 1}
            className="px-md py-xs rounded border border-brand-muted text-label disabled:opacity-50"
          >
            {t('common.previous')}
          </button>
          <span className="text-label py-xs">{page} / {data.pagination.pages}</span>
          <button
            onClick={() => setPage(p => Math.min(data.pagination.pages, p + 1))}
            disabled={page >= data.pagination.pages}
            className="px-md py-xs rounded border border-brand-muted text-label disabled:opacity-50"
          >
            {t('common.next')}
          </button>
        </div>
      )}
    </div>
  );
}
```
